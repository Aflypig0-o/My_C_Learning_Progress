19. 判断闰年
#include <stdio.h>

int isleapyear(int year)
{
    if((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0))
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

int main()
{
    int year;
    scanf("%d",&year);
    if(isleapyear(year))
    {
        printf("%d是闰年\n",year);
    }
    else
    {
        printf("%d不是闰年\n",year);
    }
    return 0;
}


20. 打印乘法表
#include <stdio.h>
int main()
{
    int number;
    scanf("%d",&number);
    for(int i=1;i<=number;++i)
    {
        for(int j=1;j<=i;++j)
        {
            printf("%d*%d=%d ",i,j,(j*i));
        }
        printf("\n");
    }
    return 0;
}


21. 猜数字游戏
#include <stdio.h>
#include <stdbool.h>

int main()
{
    bool guess_success = false;
    int target = 42;
    int guess_number;
    for(int i=0;i<5;++i)
    {
        scanf("%d",&guess_number);
        if(guess_number > target)
        {
            printf("太大了");
        }

        else if(guess_number < target)
        {
            printf("太小了");
        }

        else
        {
            printf("猜对了");
            guess_success = true;
            break;
        }
    }
    if(!guess_success)
    {
        printf("游戏结束,正确答案是%d",target);
    }
    return 0;
}
